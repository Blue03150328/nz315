## 第1步：宝塔终端只读体检

宝塔面板→终端，在同一窗口整段粘贴下方命令。括号使失败退出当前命令段，不关闭宝塔终端；不读取输出.env内容、不输出PM2全部环境，不修改数据库。

```bash
(
set -euo pipefail
cd /var/www/nz315

echo '=== 现役版本 ==='
if [ -f .deploy-version ]; then cat .deploy-version; else echo '没有.deploy-version'; fi
if [ -d .git ]; then git rev-parse HEAD; git status --short; fi
sha256sum package.json package-lock.json scripts/db-init.mjs

echo '=== 公网指纹 ==='
curl -fsS --max-time 20 https://www.nz315.cn/_nuxt/builds/latest.json
echo

echo '=== 运行环境 ==='
test -x /usr/local/node22/bin/node
/usr/local/node22/bin/node --version
command -v mysqldump || true
if [ -x /www/server/mysql80/bin/mysqldump ]; then echo '/www/server/mysql80/bin/mysqldump 可执行'; fi
df -h /var/www
free -m

echo '=== EXE附件 ==='
if [ -f public/tools/nz315-qr-tool-v1.2.0.exe ]; then sha256sum public/tools/nz315-qr-tool-v1.2.0.exe; else echo 'EXE附件缺失'; fi

/usr/local/node22/bin/node --input-type=module <<'NODE'
import fs from 'node:fs'
import { execFileSync } from 'node:child_process'
import mysql from 'mysql2/promise'
const env = {}
for (const line of fs.readFileSync('.env', 'utf8').split(/\r?\n/)) {
  const m = line.match(/^\s*([A-Z0-9_]+)\s*=\s*(.*?)\s*$/i)
  if (m) env[m[1]] = m[2].replace(/^['"]|['"]$/g, '')
}
for (const key of ['DB_HOST', 'DB_PORT', 'DB_USER', 'DB_PASSWORD', 'DB_NAME']) {
  if (!Object.hasOwn(env, key)) throw new Error('缺少配置键：' + key)
}
if (env.DB_NAME !== 'nz315') throw new Error('数据库名称不是nz315，停止体检')
const processes = JSON.parse(execFileSync('pm2', ['jlist'], { encoding: 'utf8', maxBuffer: 8 * 1024 * 1024 }))
console.log('=== PM2（仅显示运行元数据）===')
for (const name of ['nz315', 'cynx']) {
  const p = processes.find(item => item.name === name)
  console.log(JSON.stringify(p ? { name, status: p.pm2_env.status, cwd: p.pm2_env.pm_cwd, script: p.pm2_env.pm_exec_path, interpreter: p.pm2_env.exec_interpreter, port: p.pm2_env.PORT, pid: p.pid } : { name, status: '未找到' }))
}
let db
try {
  db = await mysql.createConnection({ host: env.DB_HOST, port: Number(env.DB_PORT), user: env.DB_USER, password: env.DB_PASSWORD, database: env.DB_NAME })
  console.log('=== 数据库结构（只读，无账号和业务数据）===')
  console.log(JSON.stringify({ database: env.DB_NAME, port: Number(env.DB_PORT) }))
  const [tables] = await db.query('SELECT TABLE_NAME AS name FROM information_schema.TABLES WHERE TABLE_SCHEMA=? ORDER BY TABLE_NAME', [env.DB_NAME])
  console.log(JSON.stringify({ tables: tables.map(row => row.name) }))
  const [columns] = await db.query("SELECT TABLE_NAME AS tableName,COLUMN_NAME AS columnName,COLUMN_TYPE AS columnType FROM information_schema.COLUMNS WHERE TABLE_SCHEMA=? AND ((TABLE_NAME='farm_bill' AND COLUMN_NAME IN ('dosage','store_name')) OR (TABLE_NAME='user' AND COLUMN_NAME='session_epoch') OR (TABLE_NAME='trace_code' AND COLUMN_NAME IN ('upload_batch_id','expire_date','qc_result','outer_box_code')) OR (TABLE_NAME='product' AND COLUMN_NAME IN ('original_reg_no','original_company')) OR (TABLE_NAME='product_spec' AND COLUMN_NAME='dosage_forms')) ORDER BY TABLE_NAME,COLUMN_NAME", [env.DB_NAME])
  console.log(JSON.stringify({ columns }))
} catch (error) {
  console.error('数据库体检失败，错误代码：' + (error.code || '未知'))
  process.exitCode = 1
} finally {
  if (db) await db.end()
}
NODE
)
```
